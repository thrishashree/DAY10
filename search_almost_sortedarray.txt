class Solution {
  public:
    int findTarget(vector<int>& arr, int target) {
        // code here
        int n=arr.size();
        int start=0,end=n-1;
        
        while(start<=end){
            int mid=start+(end-start)/2;
            
            if(arr[mid]==target) return mid;
            else if(arr[mid-1]==target) return mid-1;
            else if(arr[mid+1]==target) return mid+1;
            else if(target<arr[mid]) {
                end=mid-2;
                
            }else{
                start=mid+2;
            }
        }
        return -1;
    }
};
